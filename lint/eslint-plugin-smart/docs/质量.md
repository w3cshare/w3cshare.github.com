### no-var

::: warning 支持自动修复 ✅
禁止使用 var 声明变量，应使用 let 或 const
:::

```javascript
// 🔴 错误的用法
var foo = 'bar'

// ✅ 正确的用法
const foo = 'bar'
let bar = 'baz'
```

### no-console

::: warning 不支持自动修复 ❌
生产环境下禁止使用 console，开发环境下允许
:::

```javascript
// 🔴 生产环境下错误的用法
console.log('调试信息')

// ✅ 正确的用法
// 使用日志系统或条件性使用
if (process.env.NODE_ENV !== 'production') {
  console.log('开发环境调试信息')
}
```

### no-debugger

::: warning 不支持自动修复 ❌
生产环境下禁止使用 debugger
:::

```javascript
// 🔴 生产环境下错误的用法
function someFunc() {
  debugger
  return true
}

// ✅ 正确的用法
function someFunc() {
  return true
}
```

### no-alert

::: warning 不支持自动修复 ❌
生产环境下禁止使用 alert、confirm 或 prompt
:::

```javascript
// 🔴 生产环境下错误的用法
alert('提示信息')

// ✅ 正确的用法
// 使用自定义UI组件代替
showModal('提示信息')
```

### no-unexpected-multiline

::: warning 不支持自动修复 ❌
禁止出现令人困惑的多行表达式
:::

```javascript
// 🔴 错误的用法
const foo = bar(1 || 2).toString() // 这看起来像是在调用bar函数

// ✅ 正确的用法
const foo = bar
;(1 || 2).toString()
```

### no-nested-ternary

::: warning 不支持自动修复 ❌
禁止使用嵌套的三目运算符
:::

```javascript
// 🔴 错误的用法
const foo = cond1 ? val1 : cond2 ? val2 : val3

// ✅ 正确的用法
let foo
if (cond1) {
  foo = val1
} else if (cond2) {
  foo = val2
} else {
  foo = val3
}
```

### consistent-return

::: warning 不支持自动修复 ❌
要求函数要么始终有返回值，要么始终没有返回值
:::

```javascript
// 🔴 错误的用法
function doSomething(condition) {
  if (condition) {
    return true
  }
  // 没有返回语句
}

// ✅ 正确的用法
function doSomething(condition) {
  if (condition) {
    return true
  }
  return false
}
```

### no-cond-assign

::: warning 不支持自动修复 ❌
禁止在条件语句中使用赋值操作符
:::

```javascript
// 🔴 错误的用法
if ((x = y)) {
  // ...
}

// ✅ 正确的用法
if (x === y) {
  // ...
}
```

### no-use-before-define

::: warning 不支持自动修复 ❌
禁止在变量和类定义之前使用它们
:::

```javascript
// 🔴 错误的用法
console.log(foo)
const foo = 5

// ✅ 正确的用法
const foo = 5
console.log(foo)

// 函数声明可以在使用前定义（默认允许）
doSomething()
function doSomething() {}
```

### require-yield

::: warning 不支持自动修复 ❌
禁用函数内没有yield的 generator 函数
:::

```javascript
// 🔴 错误的用法
function* foo() {
  return 10
}

// ✅ 正确的用法
function* foo() {
  yield 5
  return 10
}
```

### eqeqeq

::: warning 支持自动修复 ✅
要求使用 === 和 !== 进行比较
:::

```javascript
// 🔴 错误的用法
if (x == null) {
  // ...
}

// ✅ 正确的用法
if (x === null || x === undefined) {
  // ...
}
// 或者
if (x === null) {
  // ...
}
```

### prefer-const

::: warning 支持自动修复 ✅
声明后不再被修改的变量必须使用 const
:::

```javascript
// 🔴 错误的用法
let foo = 'bar'
// foo 没有被重新赋值

// ✅ 正确的用法
const foo = 'bar'
```

### no-const-assign

::: warning 不支持自动修复 ❌
禁止修改 const 声明的变量
:::

```javascript
// 🔴 错误的用法
const foo = 'bar'
foo = 'baz'

// ✅ 正确的用法
const foo = 'bar'
// 不修改foo
```

### comma-dangle

::: warning 支持自动修复 ✅
要求或禁止末尾逗号（多行时要求）
:::

```javascript
// 🔴 错误的用法 (多行)
const obj = {
  foo: 'foo',
  bar: 'bar', // 没有末尾逗号
}

// ✅ 正确的用法 (多行)
const obj = {
  foo: 'foo',
  bar: 'bar', // 有末尾逗号
}

// ✅ 正确的用法 (单行)
const obj = { foo: 'foo', bar: 'bar' } // 不需要末尾逗号
```

### space-before-function-paren

::: warning 支持自动修复 ✅
函数括号前的空格规则
:::

```javascript
// 🔴 错误的用法
function foo() {} // 命名函数不需要空格
const bar = function () {} // 匿名函数需要空格
const baz = async () => {} // 异步箭头函数需要空格

// ✅ 正确的用法
function foo() {}
const bar = function () {}
const baz = async () => {}
```

### no-mixed-operators

::: warning 不支持自动修复 ❌
混合使用不同的运算符时需要括号明确优先级
:::

```javascript
// 🔴 错误的用法
const foo = (a && b) || c

// ✅ 正确的用法
const foo = (a && b) || c
```

### array-callback-return

::: warning 不支持自动修复 ❌
数组方法回调需要返回值
:::

```javascript
// 🔴 错误的用法
;[1, 2, 3].map(x => {
  x * 2 // 没有返回值
})

// ✅ 正确的用法
;[1, 2, 3].map(x => {
  return x * 2
})
```

### for-direction

::: warning 不支持自动修复 ❌
防止 for 循环的方向错误导致的无限循环
:::

```javascript
// 🔴 错误的用法
for (let i = 0; i < 10; i--) {
  // 会无限循环
}

// ✅ 正确的用法
for (let i = 0; i < 10; i++) {
  // 循环方向正确
}
```

### guard-for-in

::: warning 不支持自动修复 ❌
for-in 循环需要 hasOwnProperty 检查
:::

```javascript
// 🔴 错误的用法
for (const key in obj) {
  console.log(obj[key])
}

// ✅ 正确的用法
for (const key in obj) {
  if (Object.prototype.hasOwnProperty.call(obj, key)) {
    console.log(obj[key])
  }
}
```

### no-eval

::: warning 不支持自动修复 ❌
禁止使用 eval()
:::

```javascript
// 🔴 错误的用法
eval('const x = 5;')

// ✅ 正确的用法
// 使用其他方法，如 Function 构造函数或 JSON.parse
```

### no-implied-eval

::: warning 不支持自动修复 ❌
禁止使用隐式的 eval() 函数
:::

```javascript
// 🔴 错误的用法
setTimeout("alert('Hello world!')", 100)

// ✅ 正确的用法
setTimeout(() => {
  alert('Hello world!')
}, 100)
```

### no-new-func

::: warning 不支持自动修复 ❌
不要使用函数构造器创建函数
:::

```javascript
// 🔴 错误的用法
const sum = new Function('a', 'b', 'return a + b')

// ✅ 正确的用法
const sum = (a, b) => a + b
```

### complexity

::: warning 不支持自动修复 ❌
圈复杂度不超过10
:::

```javascript
// 🔴 错误的用法 - 圈复杂度过高
function doSomething() {
  if (cond1) {
    /* ... */
  } else if (cond2) {
    /* ... */
  } else if (cond3) {
    /* ... */
  } else if (cond4) {
    /* ... */
  } else if (cond5) {
    /* ... */
  }

  for (let i = 0; i < 10; i++) {
    if (arr[i] > 5) {
      /* ... */
    } else {
      /* ... */
    }
  }

  switch (value) {
    case 1:
      /* ... */ break
    case 2:
      /* ... */ break
    case 3:
      /* ... */ break
    default: /* ... */
  }
}

// ✅ 正确的用法 - 拆分为多个函数，降低复杂度
function handleConditions() {
  if (cond1) return result1
  if (cond2) return result2
  if (cond3) return result3
  return defaultResult
}

function processArray(arr) {
  return arr.map(item => (item > 5 ? doThis() : doThat()))
}

function doSomething() {
  const result1 = handleConditions()
  const result2 = processArray(someArray)
  return combineResults(result1, result2)
}
```

### no-continue

::: warning 不支持自动修复 ❌
不要使用 continue 语句
:::

```javascript
// 🔴 错误的用法
for (let i = 0; i < 10; i++) {
  if (i % 2 === 0) {
    continue
  }
  console.log(i)
}

// ✅ 正确的用法
for (let i = 0; i < 10; i++) {
  if (i % 2 !== 0) {
    console.log(i)
  }
}
```

### no-return-await

::: warning 支持自动修复 ✅
禁用不必要的 return await
:::

```javascript
// 🔴 错误的用法
async function foo() {
  return await bar()
}

// ✅ 正确的用法
async function foo() {
  return bar()
}

// 但在 try/catch 块中是有用的
async function foo() {
  try {
    return await bar()
  } catch (error) {
    // 处理错误
  }
}
```

### accessor-pairs

::: warning 不支持自动修复 ❌
getter 和 setter 应该成对出现在对象中
:::

```javascript
// 🔴 错误的用法
const obj = {
  set name(value) {
    this._name = value
  },
  // 没有对应的 getter
}

// ✅ 正确的用法
const obj = {
  set name(value) {
    this._name = value
  },
  get name() {
    return this._name
  },
}
```

### max-depth

::: warning 不支持自动修复 ❌
块语句的最大可嵌套深度不要超过4层
:::

```javascript
// 🔴 错误的用法
if (condition1) {
  if (condition2) {
    if (condition3) {
      if (condition4) {
        if (condition5) {
          // 嵌套太深
        }
      }
    }
  }
}

// ✅ 正确的用法
// 提取函数或使用提前返回来减少嵌套
if (!condition1) return
if (!condition2) return
if (!condition3) return
// 处理深度较浅的情况
```

### max-nested-callbacks

::: warning 不支持自动修复 ❌
函数的最大嵌套深度不要超过4层
:::

```javascript
// 🔴 错误的用法
doSomething(function () {
  doSomethingElse(function () {
    moreNesting(function () {
      evenMoreNesting(function () {
        tooMuchNesting(function () {
          // 嵌套太深
        })
      })
    })
  })
})

// ✅ 正确的用法
// 使用Promise链、async/await或命名函数
async function handleOperation() {
  await doSomething()
  await doSomethingElse()
  await moreNesting()
  await evenMoreNesting()
}
```

### no-negated-condition

::: warning 不支持自动修复 ❌
不要在复杂的条件表达式前加个否定操作符
:::

```javascript
// 🔴 错误的用法
if (!condition) {
  doSomething()
} else {
```

# ESLint 规则质量保证

## 命名规范和模块格式

在ESLint v9及以上版本中，配置文件的命名和格式需要与项目的模块系统保持一致。正确的配置文件命名可以确保ESLint正确加载和解析配置，从而提供更可靠的代码质量保证。

### 配置文件命名规范

| 项目类型 | 推荐文件名 | 模块系统 |
| ------- | ---------- | ------- |
| CommonJS项目 | `eslint.config.cjs` | CommonJS |
| ESModule项目 | `eslint.config.mjs` | ESModule |
| 传统配置(ESLint v8及以下) | `.eslintrc.js`或`.eslintrc.cjs` | CommonJS |

选择正确的文件扩展名不仅能确保ESLint正确识别配置文件，还能提示开发者在配置中使用正确的导入/导出语法：

- `.cjs`文件使用CommonJS的`require`和`module.exports`
- `.mjs`文件使用ESModule的`import`和`export`

### 配置示例

CommonJS配置文件 (`eslint.config.cjs`):

```js
const smartPlugin = require('eslint-plugin-smart');

module.exports = {
  extends: ['plugin:smart/recommended']
};
```

ESModule配置文件 (`eslint.config.mjs`):

```js
import smartPlugin from 'eslint-plugin-smart';

export default {
  extends: ['plugin:smart/recommended']
};
```
